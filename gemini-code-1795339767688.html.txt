    // Coordinate Helpers
    function parseCoord(str) {
      const m = str.match(/(\d{1,3})\|(\d{1,3})/);
      if (!m) return null;
      return { x: parseInt(m[1], 10), y: parseInt(m[2], 10), str: `${m[1]}|${m[2]}` };
    }

    function calcDistance(c1, c2) {
      return Math.hypot(c2.x - c1.x, c2.y - c1.y);
    }

    function getUnitBaseMinutes(unitKey) {
      const el = document.getElementById('u_' + unitKey);
      return el ? parseFloat(el.value) : 30;
    }

    function calcTravelSeconds(c1, c2, unitKey) {
      const dist = calcDistance(c1, c2);
      const baseMin = getUnitBaseMinutes(unitKey);
      const wSpeed = parseFloat(document.getElementById('worldSpeed').value) || 1;
      const uSpeed = parseFloat(document.getElementById('unitSpeed').value) || 1;
      return Math.round(dist * baseMin * (60 / (wSpeed * uSpeed)));
    }

    function formatDuration(sec) {
      const h = Math.floor(sec / 3600);
      const m = Math.floor((sec % 3600) / 60);
      const s = sec % 60;
      return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    }

    function formatTimestamp(date) {
      const pad = n => String(n).padStart(2, '0');
      return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())} ${pad(date.getHours())}:${pad(date.getMinutes())}:${pad(date.getSeconds())}`;
    }

    function isNightBonus(date) {
      const h = date.getHours();
      const start = parseInt(document.getElementById('nbStart').value, 10);
      const end = parseInt(document.getElementById('nbEnd').value, 10);
      if (start < end) return h >= start && h < end;
      if (start > end) return h >= start || h < end;
      return false;
    }

    // Main Planner Solver
    function generatePlan() {
      const rawSources = document.getElementById('sourceCoords').value.split('\n').map(s => s.trim()).filter(Boolean);
      const rawTargets = document.getElementById('targetCoords').value.split('\n').map(s => s.trim()).filter(Boolean);
      const globalArrival = new Date(document.getElementById('arrivalTime').value);
      const unitKey = document.getElementById('planUnit').value;
      const jitterEnabled = document.getElementById('applyEntropyJitter').checked;
      const minGap = parseInt(document.getElementById('minSendGap').value, 10) || 2;

      if (isNaN(globalArrival.getTime())) {
        alert('Please set a valid Target Arrival Time.');
        return;
      }

      let sources = [];
      rawSources.forEach(line => {
        const nobleMatch = line.match(/(\d+)\s*nobles?\s*in\s*(\d{1,3}\|\d{1,3})/i);
        const c = parseCoord(line);
        if (nobleMatch) {
          const count = parseInt(nobleMatch[1], 10);
          const coord = parseCoord(nobleMatch[2]);
          for (let i = 0; i < count; i++) sources.push({ ...coord, used: false });
        } else if (c) {
          sources.push({ ...c, used: false });
        }
      });

      if (sources.length === 0 || rawTargets.length === 0) {
        alert('Enter at least one valid source and target coordinate (format: 500|500).');
        return;
      }

      const commands = [];

      if (currentPlannerMode === 'edelspam') {
        rawTargets.forEach(tLine => {
          const tCoord = parseCoord(tLine);
          if (!tCoord) return;
          const cntMatch = tLine.replace(tCoord.str, '').match(/(\d+)/);
          const reqNobles = cntMatch ? parseInt(cntMatch[1], 10) : 1;

          for (let n = 0; n < reqNobles; n++) {
            const avail = sources.filter(s => !s.used);
            if (avail.length === 0) break;
            avail.sort((a, b) => calcDistance(a, tCoord) - calcDistance(b, tCoord));
            const chosen = avail[0];
            chosen.used = true;

            const arrival = new Date(globalArrival.getTime() + n * 250);
            const travelSec = calcTravelSeconds(chosen, tCoord, 'snob');
            const launch = new Date(arrival.getTime() - travelSec * 1000);
            commands.push({
              type: 'NOBLE',
              css: 'noble',
              source: chosen.str,
              target: tCoord.str,
              unit: 'snob',
              travelSec,
              launch,
              arrival
            });
          }
        });
      } else if (currentPlannerMode === 'cmdgen') {
        rawTargets.forEach(tLine => {
          const parts = tLine.split(',');
          const tCoord = parseCoord(parts[0]);
          if (!tCoord) return;
          const count = parts[1] ? parseInt(parts[1], 10) : 1;
          const arr = parts[2] ? new Date(parts[2].trim().replace(' ', 'T')) : globalArrival;

          for (let i = 0; i < count; i++) {
            const avail = sources.filter(s => !s.used);
            if (avail.length === 0) break;
            avail.sort((a, b) => calcDistance(a, tCoord) - calcDistance(b, tCoord));
            const chosen = avail[0];
            chosen.used = true;

            const travelSec = calcTravelSeconds(chosen, tCoord, unitKey);
            const launch = new Date(arr.getTime() - travelSec * 1000);
            commands.push({
              type: unitKey === 'snob' ? 'NOBLE' : 'CLEAR',
              css: unitKey === 'snob' ? 'noble' : 'clear',
              source: chosen.str,
              target: tCoord.str,
              unit: unitKey,
              travelSec,
              launch,
              arrival: arr
            });
          }
        });
      } else if (currentPlannerMode === 'advcmd') {
        const clears = parseInt(document.getElementById('clearsPerTarget').value, 10) || 0;
        const fakes = parseInt(document.getElementById('fakesPerTarget').value, 10) || 0;
        const fakeUnit = document.getElementById('fakeUnit').value;

        rawTargets.forEach(tLine => {
          const tCoord = parseCoord(tLine);
          if (!tCoord) return;

          for (let c = 0; c < clears; c++) {
            const avail = sources.filter(s => !s.used);
            if (!avail.length) break;
            avail.sort((a, b) => calcDistance(a, tCoord) - calcDistance(b, tCoord));
            const chosen = avail[0];
            chosen.used = true;
            const travelSec = calcTravelSeconds(chosen, tCoord, unitKey);
            commands.push({
              type: 'CLEAR',
              css: 'clear',
              source: chosen.str,
              target: tCoord.str,
              unit: unitKey,
              travelSec,
              launch: new Date(globalArrival.getTime() - travelSec * 1000),
              arrival: globalArrival
            });
          }

          for (let f = 0; f < fakes; f++) {
            const avail = sources.filter(s => !s.used);
            if (!avail.length) break;
            const chosen = avail[Math.floor(Math.random() * avail.length)];
            chosen.used = true;
            const travelSec = calcTravelSeconds(chosen, tCoord, fakeUnit);
            commands.push({
              type: 'FAKE',
              css: 'fake',
              source: chosen.str,
              target: tCoord.str,
              unit: fakeUnit,
              travelSec,
              launch: new Date(globalArrival.getTime() - travelSec * 1000),
              arrival: globalArrival
            });
          }
        });
      } else {
        // Simple & Advanced
        const order = document.getElementById('assignOrder').value;
        const earliest = currentPlannerMode === 'advanced' ? new Date(document.getElementById('earliestLaunch').value) : null;

        rawTargets.forEach(tLine => {
          const tCoord = parseCoord(tLine);
          if (!tCoord) return;
          let avail = sources.filter(s => !s.used);
          if (!avail.length) return;

          avail.sort((a, b) => order === 'furthest'
            ? calcDistance(b, tCoord) - calcDistance(a, tCoord)
            : calcDistance(a, tCoord) - calcDistance(b, tCoord));

          for (const candidate of avail) {
            const travelSec = calcTravelSeconds(candidate, tCoord, unitKey);
            const launch = new Date(globalArrival.getTime() - travelSec * 1000);
            if (earliest && !isNaN(earliest.getTime()) && launch < earliest) continue;
            candidate.used = true;
            commands.push({
              type: unitKey === 'snob' ? 'NOBLE' : 'CLEAR',
              css: unitKey === 'snob' ? 'noble' : 'clear',
              source: candidate.str,
              target: tCoord.str,
              unit: unitKey,
              travelSec,
              launch,
              arrival: globalArrival
            });
            break;
          }
        });
      }

      // Apply AI Stochastic Jitter if enabled
      if (jitterEnabled) {
        commands.forEach(cmd => {
          const offsetMs = (Math.floor(Math.random() * 7) - 3) * 1000 + Math.floor(Math.random() * 800);
          cmd.launch = new Date(cmd.launch.getTime() + offsetMs);
          cmd.arrival = new Date(cmd.arrival.getTime() + offsetMs);
        });
      }

      // Sort chronologically by Launch Time and enforce human send gap
      commands.sort((a, b) => a.launch - b.launch);
      for (let i = 1; i < commands.length; i++) {
        const diffSec = (commands[i].launch - commands[i - 1].launch) / 1000;
        if (diffSec < minGap) {
          const shiftMs = (minGap - diffSec) * 1000;
          commands[i].launch = new Date(commands[i].launch.getTime() + shiftMs);
          commands[i].arrival = new Date(commands[i].arrival.getTime() + shiftMs);
        }
      }

      generatedCommands = commands;
      renderSchedule();
      buildBBCode();
    }

      let heavy = parseInt(document.getElementById('simHeavy').value, 10) || 0;
      const nukeType = document.getElementById('simNukeType').value;
      const nbMult = parseFloat(document.getElementById('simNightBonus').value) || 1;

      const templates = {
        standard: { offPower: 6500 * 40 + 3000 * 130, rams: 300 },
        wallbreaker: { offPower: 5500 * 40 + 2800 * 130, rams: 500 },
        heavyoff: { offPower: 7500 * 40 + 3200 * 130, rams: 250 }
      };
      const nuke = templates[nukeType];

      let html = `<table><thead><tr><th>Wave</th><th>Wall Start→End</th><th>Def Remaining</th><th>Status</th></tr></thead><tbody>`;
      let breachWave = null;

      for (let w = 1; w <= waves; w++) {
        const startWall = wall;
        // Pre-combat ram wall drop approximation
        const defPowerBefore = (spear * 30 + sword * 40 + heavy * 150) * Math.pow(1.037, wall) * nbMult;
        const ramEffectiveness = Math.min(1, (nuke.offPower / Math.max(1, defPowerBefore)) * 1.35);
        const wallDrop = Math.min(wall, Math.max(1, Math.round((nuke.rams / 18) * ramEffectiveness)));
        const effectiveBattleWall = Math.max(0, wall - Math.ceil(wallDrop / 2));

        const defPower = Math.max(1, (spear * 30 + sword * 40 + heavy * 150) * Math.pow(1.037, effectiveBattleWall) * nbMult);
        const ratio = nuke.offPower / defPower;

        if (ratio >= 1) {
          spear = 0; sword = 0; heavy = 0;
          wall = Math.max(0, wall - wallDrop * 2);
          if (!breachWave) breachWave = w;
        } else {
          const lossFraction = Math.pow(ratio, 1.5);
          spear = Math.max(0, Math.round(spear * (1 - lossFraction)));
          sword = Math.max(0, Math.round(sword * (1 - lossFraction)));
          heavy = Math.max(0, Math.round(heavy * (1 - lossFraction)));
          wall = Math.max(0, wall - wallDrop);
        }

        const totalTroops = spear + sword + heavy;
        html += `<tr>
          <td><strong>#${w}</strong></td>
          <td>Lv.${startWall} ➔ Lv.${wall}</td>
          <td>${totalTroops.toLocaleString()}</td>
          <td>${totalTroops === 0 ? '<span class="badge success">CLEARED</span>' : '<span class="badge danger">HOLDING</span>'}</td>
        </tr>`;
      }
      html += `</tbody></table>`;

      const summary = breachWave
        ? `<div class="alert-box success"><strong>AI Verdict:</strong> Stack breaks on <strong>Wave #${breachWave}</strong>. Schedule noble train immediately behind Wave #${breachWave}.</div>`
        : `<div class="alert-box warn"><strong>AI Verdict:</strong> Stack survives all ${waves} waves (Wall ends at Lv.${wall}). Add more wallbreaker nukes or pivot targets.</div>`;

      document.getElementById('simResults').innerHTML = summary + html;
    }

    // AI WAR ROOM 2: TARGET CLUSTER & CENTRALITY SCORER
    function scoreTargetClusters() {
      const lines = document.getElementById('clusterInput').value.split('\n');
      const coords = [];
      lines.forEach(l => {
        const c = parseCoord(l);
        if (c && !coords.some(x => x.str === c.str)) coords.push(c);
      });

      if (coords.length < 2) {
        alert('Enter at least 2 distinct coordinates to analyze cluster centrality.');
        return;
      }

      const scored = coords.map(c => {
        let supportScore = 0;
        let neighborsWithin5 = 0;
        coords.forEach(other => {
          if (c.str === other.str) return;
          const d = calcDistance(c, other);
          supportScore += 1 / Math.max(0.5, d);
          if (d <= 5) neighborsWithin5++;
        });
        return { ...c, supportScore, neighborsWithin5 };
      });

      scored.sort((a, b) => b.supportScore - a.supportScore);
      const maxScore = scored[0].supportScore || 1;

      let html = `<table><thead><tr><th>Rank</th><th>Coord</th><th>Neighbors (&le;5f)</th><th>Leverage</th></tr></thead><tbody>`;
      scored.forEach((item, idx) => {
        const pct = Math.round((item.supportScore / maxScore) * 100);
        html += `<tr>
          <td>#${idx + 1}</td>
          <td><strong>${item.str}</strong></td>
          <td>${item.neighborsWithin5}</td>
          <td><span class="badge ${idx === 0 ? 'success' : ''}">${pct}% Anchor</span></td>
        </tr>`;
      });
      html += `</tbody></table>`;
      document.getElementById('clusterResults').innerHTML = html;
    }

    // AI WAR ROOM 3: SNIPE & BACKTIME CALCULATOR
    function calcBacktimeAndSnipe() {
      const orig = parseCoord(document.getElementById('btOrigin').value);
      const targ = parseCoord(document.getElementById('btTarget').value);
      const unit = document.getElementById('btEnemyUnit').value;
      const impact = new Date(document.getElementById('btImpactTime').value);

      if (!orig || !targ || isNaN(impact.getTime())) {
        alert('Enter valid coordinates and impact timestamp.');
        return;
      }

      const travelSec = calcTravelSeconds(orig, targ, unit);
      const enemyLaunch = new Date(impact.getTime() - travelSec * 1000);
      const enemyReturn = new Date(impact.getTime() + travelSec * 1000);
      const cancelMidpoint = new Date(Date.now() + (impact.getTime() - Date.now()) / 2);

      document.getElementById('btResults').innerHTML = `
        <div class="alert-box">
          <strong>Distance:</strong> ${calcDistance(orig, targ).toFixed(2)} fields (${formatDuration(travelSec)})<br>
          <strong>Enemy Sent At:</strong> ${formatTimestamp(enemyLaunch)}<br>
          <strong>Enemy Troops Return Home:</strong> <span style="color:#f59e0b;font-weight:700;">${formatTimestamp(enemyReturn)}</span><br>
          <strong>Self-Cancel Snipe Midpoint (from now):</strong> <span style="color:#10b981;font-weight:700;">${formatTimestamp(cancelMidpoint)}</span>
        </div>
      `;
    }
